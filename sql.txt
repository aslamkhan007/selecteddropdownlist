
ALTER PROCEDURE [dbo].[sp_NewRequestReceivedForAdmin]    
(
@LoginId nvarchar(50),
@Division nvarchar(50) = NULL,
@InstallationType  nvarchar(50) = NULL,
 @SearchTxt NVARCHAR(100) = NULL

)	
AS 
BEGIN 
Declare @Role varchar(20)
Select @Role=[Role] From tbl_LoginMaster Where UserID=@LoginId
Declare @DivisionCode varchar(10)
Select @DivisionCode=Area From tbl_CEIAreaCovered Where HeadOffice=@Division
    
SELECT distinct
        I.Id AS InspectionId, 
        I.InstallationType AS Installationfor, 
        I.ApplicantType AS ApplicantFor,
		I.AssignTo,
		I.CreatedBy as CreatedBy,
        --WI.NameOfOwner AS OwnerName, 
			Case When (WI.NameOfOwner is NULL AND WI.NameOfAgency is NULL)
			Then (Select Case When r.NameOfAgency is not Null then r.NameOfAgency Else r.NameOfOwner End From tbl_RegistrationSiteOwner as r Where r.PANNumber=I.Createdby)
			Else
			CASE
			-- WHEN     WI.NameOfOwner != NULL OR  WI.NameOfOwner !='' THEN  WI.NameOfOwner     
			-- ELSE   WI.NameOfAgency END 
            WHEN     WI.NameOfAgency != NULL OR  WI.NameOfAgency !='' THEN  WI.NameOfAgency     
			ELSE   WI.NameOfOwner END 
			End
			As OwnerName, 

			CD.FirmName AS ContractorName,
dbo.GetTotalPendingDays (I.Id) AS PendingDays,
CONVERT(varchar, I.CreatedDate, 103) AS RequestDate,

CASE WHEN I.ApplicationStatus = 'Submit' THEN 'New' ELSE 'OtherStatus' END AS ApplicationStatus,
I.TypeOfInspection
FROM 
    tbl_Inspection AS I    
--LEFT JOIN 
--    tbl_StaffLogin AS SL ON I.AssignTo = SL.StaffUserId      
LEFT JOIN 
    tbl_WorkIntimation AS WI ON WI.Id = I.IntimationId  
LEFT JOIN 
    tbl_ContractorDetails AS CD ON WI.Createdby = CD.UserId
WHERE 
    (I.ApplicationStatus = 'Submit' ) AND @Role='Admin' 
	AND (@DivisionCode IS NULL OR @DivisionCode = '0' OR I.Division = @DivisionCode)
	AND (I.InspectionNewOrExist!='Existing' or I.InspectionNewOrExist is null)  
	  AND (
            (@InstallationType = '1' AND I.InstallationType IN ('Lift', 'Escalator','Lift/Escalator','MultiLift','MultiEscalator')) OR  
            (@InstallationType = '0' AND I.InstallationType IN ('Line', 'Generating Set','Substation Transformer','Multiple')) OR  
            (@InstallationType IS NULL)  
        ) AND I.Active = 1  
		AND (
            @SearchTxt IS NULL OR
            CAST(I.Id AS NVARCHAR) LIKE '%' + @SearchTxt + '%' OR
            I.InstallationType LIKE '%' + @SearchTxt + '%' OR
            I.ApplicantType LIKE '%' + @SearchTxt + '%' OR
            I.TypeOfInspection LIKE '%' + @SearchTxt + '%' OR
            WI.District LIKE '%' + @SearchTxt + '%' OR
            I.Division LIKE '%' + @SearchTxt + '%' OR
            CD.FirmName LIKE '%' + @SearchTxt + '%' OR
            WI.NameOfOwner LIKE '%' + @SearchTxt + '%' OR
            WI.NameOfAgency LIKE '%' + @SearchTxt + '%' OR
            CONVERT(VARCHAR, I.CreatedDate, 103) LIKE '%' + @SearchTxt + '%' OR
            CONVERT(VARCHAR, I.ApprovedDate, 103) LIKE '%' + @SearchTxt + '%' OR
            I.ApplicationStatus LIKE '%' + @SearchTxt + '%'
        ) 
	Order By PendingDays desc
END



go
